#include <iostream>
#include <unordered_map>
#include <memory>

class LRUCache {
private:
    struct Node { int key, val; ... };
    std::unordered_map<int, Node*> cache;

public:
    int get(int key) {
        if (cache.find(key) == cache.end()) return -1;
        moveToHead(cache[key]);
        return cache[key]->val;
    }
};

Can you explain std::unique_ptr vs std::shared_ptr and RAII memory safety in C++?
Excellent question! std::unique_ptr models exclusive ownership with zero overhead (move-only):
auto ptr = std::make_unique<Resource>();
// Exclusive ownership; automatically freed upon scope exit
In contrast, std::shared_ptr maintains an atomic reference-counted control block. Prefer std::unique_ptr by default!
